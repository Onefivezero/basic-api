package main

import (
	"bytes"
	"encoding/json"
	"fmt"
	"reflect"
)

func parseList(val []map[string]any, responseType reflect.Type) {
	for _, i := range val {
		fmt.Printf("pL: %v", i)
		fmt.Printf("pL: %v", responseType)
		parseMap(i, responseType)
	}
}

func parseMap(map_ map[string]any, T reflect.Type) {
	for x, i := range reflect.VisibleFields(T) {
		fmt.Println(x, ",", i.Type)
		mapFieldValue, ok := map_[i.Name]
		if !ok {
			continue
		}
		structType := i.Type
		structKind := structType.Kind()

		if structKind == reflect.Array || structKind == reflect.Slice {
			fieldValAsList, ok := mapFieldValue.([]any)
			fmt.Println("List.")
			otherConvertedArr := make([]map[string]any, 0)
			for _, i := range fieldValAsList {
				otherConvertedArr = append(otherConvertedArr, i.(map[string]any))
			}
			if !ok {
				fmt.Printf("Skip: %s\n", reflect.TypeOf(mapFieldValue))
				fmt.Printf("Reason: %v\n", mapFieldValue)
				continue
			}
			parseList(otherConvertedArr, structType)
		} else if structKind == reflect.Struct {
			convertedMap, ok := mapFieldValue.(map[string]any)
			if !ok {
				fmt.Printf("Skip: %s\n", i.Name)
				fmt.Printf("Reason: %v\n", mapFieldValue)
				continue
			}
			parseMap(convertedMap, structType)
		} else if structType.AssignableTo(reflect.TypeOf(mapFieldValue)) {
			fmt.Println("assignable.")
		} else {
			fmt.Println("Noo.")
		}

	}
}

func Verify(data []byte, T reflect.Type) {
	decoder := json.NewDecoder(bytes.NewReader(data))

	switch data[0] {
	case '[':
		listVal := make([]map[string]any, 0)
		err := decoder.Decode(&listVal)
		if err != nil {
			fmt.Println("LISTVAL NOT VALID")
			return
		}
		parseList(listVal, T)
	case '{':
		mapVal := make(map[string]any)
		err := decoder.Decode(&mapVal)
		if err != nil {
			fmt.Println("MAPVAL NOT VALID")
			fmt.Println(err)
			return
		}
		parseMap(mapVal, T)
	default:
		fmt.Printf("Not valid: %c", data[0])
		return
	}
}
